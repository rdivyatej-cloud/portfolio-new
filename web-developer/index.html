// Idempotent Metering Record Pipeline
app.post('/api/v1/meter/event', idempotencyMiddleware, async (req, res) => {
  const { tenantId, eventId, tokenUsage, modelTier } = req.body;
  const cost = calculateUsageRate(modelTier, tokenUsage);
  await db.transaction(async (trx) => {
    await quotaService.verifyAndDeduct(trx, tenantId, cost);
    await billingService.recordLedger(trx, {
      eventId, tenantId, cost, timestamp: new Date()
    });
  });
  return res.status(200).json({ status: 'recorded', cost });
});
