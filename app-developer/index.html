// Token Cost Tier Calculation
function calculateUsageRate(tier, { cached, reasoning, output }) {
  const rates = PRICING_TIERS[tier] || PRICING_TIERS.standard;
  return (
    (cached * rates.cachedPerK / 1000) +
    (reasoning * rates.reasoningPerK / 1000) +
    (output * rates.outputPerK / 1000)
  );
}
