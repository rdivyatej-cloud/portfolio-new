// Normalized Monthly Spend Calculation
export const computeNormalizedSpend = (subscriptions) => {
  return subscriptions.reduce((total, sub) => {
    if (!sub.active) return total;
    const monthlyRate = sub.billingCycle === 'yearly' 
      ? sub.cost / 12 
      : sub.cost;
    return total + monthlyRate;
  }, 0);
};

// Real-Time Quota Gauge Indicator
export function QuotaGauge({ currentUsage, maxQuota }) {
  const percentage = Math.min(100, Math.round((currentUsage / maxQuota) * 100));
  const isNearLimit = percentage >= 85;
  return (
    <div className="gauge-container">
      <div className="gauge-bar" style={{ width: `${percentage}%`, background: isNearLimit ? '#ef4444' : '#00e5ff' }} />
      <span className="gauge-label">{percentage}% Quota Consumed</span>
    </div>
  );
}
