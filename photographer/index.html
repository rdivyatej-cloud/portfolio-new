class WorkoutPlan:
    def __init__(self, name: str, exercises: list):
        self.name = name
        self.exercises = exercises  # List of (exercise_name, sets, reps, cal_per_rep)

    def calculate_total_calories(self, index=0) -> float:
        # Recursive calorie burn aggregation
        if index >= len(self.exercises):
            return 0.0
        _, sets, reps, cal = self.exercises[index]
        return (sets * reps * cal) + self.calculate_total_calories(index + 1)

// Dynamic Programming: Optimal Substructure Memoization
int maxProfitWithCooldown(vector<int>& prices) {
    int sold = 0, rest = 0, hold = INT_MIN;
    for (int p : prices) {
        int prev_sold = sold;
        sold = hold + p;
        hold = max(hold, rest - p);
        rest = max(rest, prev_sold);
    }
    return max(sold, rest);
}

import pandas as pd
import numpy as np

def generate_analytics_summary(csv_path: str):
    df = pd.read_csv(csv_path)
    df['normalized_spend'] = np.where(df['cycle'] == 'yearly', df['cost'] / 12, df['cost'])
    breakdown = df.groupby('category')['normalized_spend'].agg(['sum', 'count', 'mean'])
    return breakdown.sort_values(by='sum', ascending=False)
